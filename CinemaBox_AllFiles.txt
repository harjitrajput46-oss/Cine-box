# CINEMA BOX

Android phone + Android TV (API 22+) streaming app. Java, XML, ExoPlayer 2, Glide.

## Build
Push this project to GitHub (branch `main`). Actions > "Build APK" runs automatically.
Download `CinemaBox-debug-apk` from the run's Artifacts. Installable file: app-debug.apk

## Content
- Sample metadata: `app/src/main/assets/movies.json` (NO stream links included).
- Real catalog: set `API_BASE_URL` in `Config.java` (https). The app calls `GET {API_BASE_URL}/catalog`
  and expects `{"items":[ ... ]}` in the same format as movies.json.
- Streams: put your licensed/authorized MP4, HLS (.m3u8) or DASH (.mpd) URL in `streamUrl`
  (movies) or in each episode `url` (series). Without it the app shows "No streaming source".
- Adding titles needs only JSON changes, no UI code changes.

## JSON fields
id, title, type (movie|series|tv|anime), language, year, rating, ratingSource, genres[], cast[],
director, runtime, synopsis, poster, backdrop, streamUrl, tags[] (hollywood, bollywood, south,
hindi_dubbed, trending, latest, popular, featured), seasons[{number, episodes[{title, description, url}]}]

    branches: [ main ]
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - uses: android-actions/setup-android@v3
      - uses: gradle/actions/setup-gradle@v3
        with:
          gradle-version: 7.6.4
      - name: Build debug APK
        run: gradle assembleDebug --no-daemon
      - uses: actions/upload-artifact@v4
        with:
          name: CinemaBox-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk

package com.cinemabox.app.model;

import org.json.JSONArray;
import org.json.JSONObject;

import java.util.ArrayList;
import java.util.List;

public class Content {
    public String id = "", title = "", type = "movie", language = "", synopsis = "", director = "",
            runtime = "", poster = "", backdrop = "", streamUrl = "", ratingSource = "";
    public int year;
    public double rating;
    public List<String> genres = new ArrayList<>();
    public List<String> cast = new ArrayList<>();
    public List<String> tags = new ArrayList<>();
    public List<Season> seasons = new ArrayList<>();

    public static class Episode {
        public String id = "", title = "", description = "", url = "";
    }

    public static class Season {
        public int number;
        public List<Episode> episodes = new ArrayList<>();
    }

    public boolean hasEpisodes() {
        return !seasons.isEmpty();
    }

    public String typeLabel() {
        switch (type) {
            case "movie": return "Movie";
            case "series": return "Series";
            case "anime": return "Anime";
            case "tv": return "TV Show";
            default: return type;
        }
    }

    public static Content fromJson(JSONObject o) {
        Content c = new Content();
        c.id = o.optString("id", "");
        c.title = o.optString("title", "Untitled");
        c.type = o.optString("type", "movie");
        c.language = o.optString("language", "");
        c.synopsis = o.optString("synopsis", "");
        c.director = o.optString("director", "");
        c.runtime = o.optString("runtime", "");
        c.poster = o.optString("poster", "");
        c.backdrop = o.optString("backdrop", "");
        c.streamUrl = o.optString("streamUrl", "");
        c.ratingSource = o.optString("ratingSource", "");
        c.year = o.optInt("year", 0);
        c.rating = o.optDouble("rating", 0);
        c.genres = strings(o.optJSONArray("genres"), false);
        c.cast = strings(o.optJSONArray("cast"), false);
        c.tags = strings(o.optJSONArray("tags"), true);
        JSONArray ss = o.optJSONArray("seasons");
        if (ss != null) {
            for (int i = 0; i < ss.length(); i++) {
                JSONObject so = ss.optJSONObject(i);
                if (so == null) continue;
                Season s = new Season();
                s.number = so.optInt("number", i + 1);
                JSONArray es = so.optJSONArray("episodes");
                if (es != null) {
                    for (int j = 0; j < es.length(); j++) {
                        JSONObject eo = es.optJSONObject(j);
                        if (eo == null) continue;
                        Episode e = new Episode();
                        e.id = c.id + "_s" + s.number + "e" + (j + 1);
                        e.title = eo.optString("title", "Episode " + (j + 1));
                        e.description = eo.optString("description", "");
                        e.url = eo.optString("url", "");
                        s.episodes.add(e);
                    }
                }
                c.seasons.add(s);
            }
        }
        return c;
    }

    private static List<String> strings(JSONArray a, boolean lower) {
        List<String> out = new ArrayList<>();
        if (a == null) return out;
        for (int i = 0; i < a.length(); i++) {
            String s = a.optString(i, "");
            if (!s.isEmpty()) out.add(lower ? s.toLowerCase() : s);
        }
        return out;
    }

    public static String join(List<String> l, String sep) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < l.size(); i++) {
            if (i > 0) sb.append(sep);
            sb.append(l.get(i));
        }
        return sb.toString();
    }

import android.content.Context;

import java.io.InputStreamReader;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.ArrayList;
import java.util.List;

public class Repository {

    public interface Callback {
        void onResult(List<Content> items, String message);
    }

    private static List<Content> cache;

    public static void load(final Context ctx, boolean force, final Callback cb) {
        if (!force && cache != null) {
            cb.onResult(cache, null);
            return;
        }
        final Context app = ctx.getApplicationContext();
        final Handler main = new Handler(Looper.getMainLooper());
        new Thread(new Runnable() {
            @Override
            public void run() {
                List<Content> list = null;
                String msg = null;
                if (!Config.API_BASE_URL.isEmpty()) {
                    try {
                        list = parse(download(Config.API_BASE_URL + Config.CATALOG_PATH));
                    } catch (Exception e) {
                        msg = "Network error. Showing offline sample data. Tap to retry.";
                    }
                }
                if (list == null) {
                    try {
                        list = parse(readAsset(app));
                    } catch (Exception e) {
                        list = new ArrayList<>();
                        msg = "Could not load catalog. Tap to retry.";
                    }
                }
                final List<Content> result = list;
                final String message = msg;
                if (message == null) cache = result;
                main.post(new Runnable() {
                    @Override
                    public void run() {
                        cb.onResult(result, message);
                    }
                });
            }
        }).start();
    }

    private static String download(String url) throws Exception {
        HttpURLConnection con = (HttpURLConnection) new URL(url).openConnection();
        con.setConnectTimeout(10000);
        con.setReadTimeout(10000);
        try {
            if (con.getResponseCode() != 200) throw new Exception("HTTP " + con.getResponseCode());
            return read(con.getInputStream());
        } finally {
            con.disconnect();
        }
    }

    private static String readAsset(Context c) throws Exception {
        return read(c.getAssets().open("movies.json"));
    }

    private static String read(InputStream in) throws Exception {
        BufferedReader r = new BufferedReader(new InputStreamReader(in, "UTF-8"));
        StringBuilder sb = new StringBuilder();
        String line;
        while ((line = r.readLine()) != null) sb.append(line).append('\n');
        r.close();
        return sb.toString();
    }

    private static List<Content> parse(String s) throws Exception {
        s = s.trim();
        JSONArray arr = s.startsWith("[") ? new JSONArray(s) : new JSONObject(s).optJSONArray("items");
        List<Content> out = new ArrayList<>();
        if (arr == null) return out;
        for (int i = 0; i < arr.length(); i++) {
            JSONObject o = arr.optJSONObject(i);
            if (o != null) out.add(Content.fromJson(o));
        }
        return out;
    }

    public static final String WATCHLIST = "watchlist";

    public static final String CONTINUE = "continue";
    public static final String RECENT = "recent";
    public static final String HISTORY = "history";

    private final SharedPreferences sp;

    public UserStore(Context c) {
        sp = c.getApplicationContext().getSharedPreferences("cinemabox", Context.MODE_PRIVATE);
    }

    public List<String> list(String key) {
        List<String> out = new ArrayList<>();
        try {
            JSONArray a = new JSONArray(sp.getString(key, "[]"));
            for (int i = 0; i < a.length(); i++) out.add(a.getString(i));
        } catch (Exception ignored) {
        }
        return out;
    }

    private void save(String key, List<String> l) {
        sp.edit().putString(key, new JSONArray(l).toString()).apply();
    }

    public boolean contains(String key, String id) {
        return list(key).contains(id);
    }

    public boolean toggle(String key, String id) {
        List<String> l = list(key);
        boolean added;
        if (l.contains(id)) {
            l.remove(id);
            added = false;
        } else {
            l.add(0, id);
            added = true;
        }
        save(key, l);
        return added;
    }

    public void addFront(String key, String id, int max) {
        List<String> l = list(key);
        l.remove(id);
        l.add(0, id);
        while (l.size() > max) l.remove(l.size() - 1);
        save(key, l);
    }

    public void remove(String key, String id) {
        List<String> l = list(key);
        l.remove(id);
        save(key, l);
    }

    public void clear(String key) {
        sp.edit().remove(key).apply();
    }

    public long getPosition(String mediaId) {
        return sp.getLong("pos_" + mediaId, 0);
    }

    public String getLast(String contentId) {
        return sp.getString("last_" + contentId, null);
    }

    public void saveProgress(String contentId, String mediaId, long pos, long dur, boolean hasEpisodes) {
        boolean done = dur > 0 && pos >= dur * 95 / 100;
        SharedPreferences.Editor e = sp.edit();
        if (done) e.remove("pos_" + mediaId);
        else if (pos > 2000) e.putLong("pos_" + mediaId, pos);
        e.putString("last_" + contentId, mediaId);
        e.apply();
        if (done && !hasEpisodes) remove(CONTINUE, contentId);
        else if (pos > 2000 || done) addFront(CONTINUE, contentId, 30);
    }

    public void clearProgress() {
        SharedPreferences.Editor e = sp.edit();
        for (Map.Entry<String, ?> en : sp.getAll().entrySet()) {
            String k = en.getKey();
            if (k.startsWith("pos_") || k.startsWith("last_")) e.remove(k);
        }
        e.remove(CONTINUE);
        e.apply();
    }

    public boolean resumeEnabled() {
        return sp.getBoolean("resume", true);
    }

    public void setResume(boolean v) {
        sp.edit().putBoolean("resume", v).apply();
    }
}

package com.cinemabox.app;

import android.graphics.drawable.Drawable;
import android.widget.ImageView;

import com.bumptech.glide.Glide;
import com.bumptech.glide.RequestBuilder;
import com.bumptech.glide.load.resource.bitmap.CenterCrop;
import com.bumptech.glide.load.resource.bitmap.RoundedCorners;

public class ImageUtil {
    /** Returns true if an image URL was loaded, false if the placeholder is shown. */
    public static boolean load(ImageView iv, String url, int radiusDp) {
        if (url == null || url.isEmpty()) {
            Glide.with(iv).clear(iv);
            iv.setImageResource(R.drawable.bg_poster);
            return false;
        }
        RequestBuilder<Drawable> rb = Glide.with(iv).load(url).placeholder(R.drawable.bg_poster);
        if (radiusDp > 0) {
            int px = (int) (radiusDp * iv.getResources().getDisplayMetrics().density);
            rb = rb.transform(new CenterCrop(), new RoundedCorners(px));
        } else {
            rb = rb.centerCrop();
        }
        rb.into(iv);
        return true;
    }
}

        addRow(r, "Continue Watching", byIds(store.list(UserStore.CONTINUE)));
        addRow(r, "Trending Movies", where(new Filter() {
            public boolean ok(Content c) { return "movie".equals(c.type) && c.tags.contains("trending"); }
        }));
        addRow(r, "Popular Movies", where(new Filter() {
            public boolean ok(Content c) { return "movie".equals(c.type) && c.tags.contains("popular"); }
        }));
        addRow(r, "Latest Movies", where(new Filter() {
            public boolean ok(Content c) { return "movie".equals(c.type) && c.tags.contains("latest"); }
        }));
        addRow(r, "Popular Web Series", where(new Filter() {
            public boolean ok(Content c) { return "series".equals(c.type) && c.tags.contains("popular"); }
        }));
        addRow(r, "Anime Collection", where(new Filter() {
            public boolean ok(Content c) { return "anime".equals(c.type); }
        }));
        addRow(r, "Hindi Dubbed", where(new Filter() {
            public boolean ok(Content c) { return c.tags.contains("hindi_dubbed"); }
        }));
        addRow(r, "Recommended For You", recommended());
        addRow(r, "Recently Watched", byIds(store.list(UserStore.RECENT)));
        return r;
    }

    private void addRow(List<RowAdapter.Row> rows, String title, List<Content> items) {
        if (!items.isEmpty()) rows.add(new RowAdapter.Row(title, items));
    }

    private List<Content> where(Filter f) {
        List<Content> out = new ArrayList<>();
        for (Content c : all) if (f.ok(c)) out.add(c);
        return out;
    }

    private Content find(String id) {
        for (Content c : all) if (c.id.equals(id)) return c;
        return null;
    }

    private List<Content> byIds(List<String> ids) {
        List<Content> out = new ArrayList<>();
        for (String id : ids) {
            Content c = find(id);
            if (c != null) out.add(c);
        }
        return out;
    }

    private List<Content> recommended() {
        final Set<String> liked = new HashSet<>();
        Set<String> seen = new HashSet<>();
        String[] keys = {UserStore.WATCHLIST, UserStore.FAVORITES, UserStore.RECENT, UserStore.CONTINUE};
        for (String k : keys) {
            for (String id : store.list(k)) {
                Content c = find(id);
                if (c != null) {
                    liked.addAll(c.genres);
                    seen.add(id);
                }
            }
        }
        List<Content> out = new ArrayList<>();
        for (Content c : all) if (!seen.contains(c.id)) out.add(c);
        Collections.sort(out, new Comparator<Content>() {
            @Override
            public int compare(Content a, Content b) {
                return Double.compare(score(b, liked), score(a, liked));
            }
        });
        return out.size() > 12 ? new ArrayList<>(out.subList(0, 12)) : out;
    }

    private double score(Content c, Set<String> liked) {
        int overlap = 0;
        for (String g : c.genres) if (liked.contains(g)) overlap++;
        return overlap * 10 + c.rating;
    }

    private boolean matchesCategory(Content c, String cat) {
        switch (cat) {
            case "Movies": return "movie".equals(c.type);
            case "Hollywood": return c.tags.contains("hollywood");
            case "Bollywood": return c.tags.contains("bollywood");
            case "South Indian": return c.tags.contains("south");
            case "Hindi Dubbed": return c.tags.contains("hindi_dubbed");
            case "Web Series": return "series".equals(c.type);
            case "TV Shows": return "tv".equals(c.type);
            case "Anime": return "anime".equals(c.type);
            case "Trending": return c.tags.contains("trending");
            case "Latest Releases": return c.tags.contains("latest");
            case "Popular": return c.tags.contains("popular");
            default: return true;
        }
    }

    private boolean matchesQuery(Content c) {
        if (c.title.toLowerCase().contains(query)) return true;
        if (c.director.toLowerCase().contains(query)) return true;
        for (String s : c.cast) if (s.toLowerCase().contains(query)) return true;
        for (String s : c.genres) if (s.toLowerCase().contains(query)) return true;
        return false;
    }

    private List<Content> filter() {
        String cat = CATEGORIES[category];
        List<Content> base;
        switch (cat) {
            case "Continue Watching": base = byIds(store.list(UserStore.CONTINUE)); break;
            case "Watchlist": base = byIds(store.list(UserStore.WATCHLIST)); break;
            case "Favorites": base = byIds(store.list(UserStore.FAVORITES)); break;
            case "Recently Watched": base = byIds(store.list(UserStore.RECENT)); break;
            default: base = all;
        }
        String lang = val(spLang), genre = val(spGenre), year = val(spYear);
        int typeIdx = spType.getSelectedItemPosition();
        String typeKey = typeIdx > 0 ? TYPE_KEYS[typeIdx] : null;
        List<Content> out = new ArrayList<>();
        for (Content c : base) {
            if (!matchesCategory(c, cat)) continue;
            if (!query.isEmpty() && !matchesQuery(c)) continue;
            if (lang != null && !lang.equalsIgnoreCase(c.language)) continue;
            if (genre != null) {
                boolean g = false;
                for (String s : c.genres) if (s.equalsIgnoreCase(genre)) g = true;
                if (!g) continue;
            }
            if (year != null && !year.equals(String.valueOf(c.year))) continue;
            if (typeKey != null && !typeKey.equals(c.type)) continue;
            out.add(c);
        }
        return out;
    }

    private void open(Content c) {
        saveHistory();
        Intent i = new Intent(this, DetailActivity.class);
        i.putExtra("id", c.id);
        startActivity(i);
    }

    private void saveHistory() {
        String q = searchBox.getText().toString().trim();
        if (!q.isEmpty()) {
            store.addFront(UserStore.HISTORY, q, 10);
            refreshHistory();
        }
    }

    private void refreshHistory() {
        searchBox.setAdapter(new ArrayAdapter<>(this, android.R.layout.simple_dropdown_item_1line,
                store.list(UserStore.HISTORY)));
    }

    private void showSettings() {
        String[] items = {
                "Resume playback: " + (store.resumeEnabled() ? "ON" : "OFF") + " (tap to toggle)",
                "Clear search history",
                "Clear continue watching",
                "Clear watchlist",
                "Clear favorites",
                "Clear recently watched",
                "About"};
        new AlertDialog.Builder(this).setTitle("Settings").setItems(items, new android.content.DialogInterface.OnClickListener() {
            @Override
            public void onClick(android.content.DialogInterface d, int which) {
                switch (which) {
                    case 0: store.setResume(!store.resumeEnabled()); break;
                    case 1: store.clear(UserStore.HISTORY); refreshHistory(); break;
                    case 2: store.clearProgress(); break;
                    case 3: store.clear(UserStore.WATCHLIST); break;
                    case 4: store.clear(UserStore.FAVORITES); break;
                    case 5: store.clear(UserStore.RECENT); break;
                    default: Toast.makeText(MainActivity.this, "CINEMA BOX 1.0", Toast.LENGTH_SHORT).show();
                }
                render();
            }
        }).show();
    }

    @Override
    public void onBackPressed() {
        if (ready && !isHome()) {
            category = 0;
            chipAdapter.notifyDataSetChanged();
            searchBox.setText("");
            spLang.setSelection(0);
            spGenre.setSelection(0);
            spYear.setSelection(0);
            spType.setSelection(0);
            query = "";
            render();
        } else {
            super.onBackPressed();
        }
    }

    private class ChipAdapter extends RecyclerView.Adapter<ChipAdapter.VH> {
        class VH extends RecyclerView.ViewHolder {

                super(v);
                t = (TextView) v;

                    category = p;
                    notifyDataSetChanged();

                }
            });
        }

        @Override

            return CATEGORIES.length;
        }
    }

    private void refresh() {
        boolean resume = store.list(UserStore.CONTINUE).contains(c.id);
        btnWatch.setText(resume ? "Resume" : "Watch Now");
        btnWatchlist.setText(store.contains(UserStore.WATCHLIST, c.id) ? "In Watchlist" : "Add to Watchlist");
        btnFavorite.setText(store.contains(UserStore.FAVORITES, c.id) ? "Favorited" : "Add to Favorites");
        if (c.hasEpisodes()) populateEpisodes();
    }

    private void populateEpisodes() {
        episodesBox.removeAllViews();
        if (seasonIdx >= c.seasons.size()) return;
        final Content.Season s = c.seasons.get(seasonIdx);
        for (int i = 0; i < s.episodes.size(); i++) {
            final int idx = i;
            Content.Episode e = s.episodes.get(i);
            View v = LayoutInflater.from(this).inflate(R.layout.item_episode, episodesBox, false);
            ((TextView) v.findViewById(R.id.epTitle)).setText((i + 1) + ". " + e.title);
            ((TextView) v.findViewById(R.id.epDesc)).setText(e.description);
            long pos = store.getPosition(e.id);
            ((TextView) v.findViewById(R.id.epStatus)).setText(pos > 0 ? "Resume at " + fmt(pos) : "");
            v.setOnClickListener(new View.OnClickListener() {
                @Override
                public void onClick(View x) {
                    play(seasonIdx, idx);
                }
            });
            episodesBox.addView(v);
        }

    private String fmt(long ms) {
        long s = ms / 1000;
        return String.format(Locale.US, "%d:%02d", s / 60, s % 60);
    }

    private void watchNow() {
        if (!c.hasEpisodes()) {
            play(-1, 0);
            return;
        }
        int s = 0, e = 0;
        String last = store.getLast(c.id);
        if (last != null) {
            for (int si = 0; si < c.seasons.size(); si++) {
                List<Content.Episode> eps = c.seasons.get(si).episodes;
                for (int ei = 0; ei < eps.size(); ei++) {
                    if (eps.get(ei).id.equals(last)) {
                        s = si;
                        e = ei;
                    }
                }
            }
        }
        play(s, e);
    }

    private void play(int season, int episode) {
        String url;
        if (season < 0) url = c.streamUrl;
        else url = c.seasons.get(season).episodes.get(episode).url;
        if (url == null || url.isEmpty()) {
            new AlertDialog.Builder(this)
                    .setTitle("No streaming source")
                    .setMessage("No authorized stream is connected for this title. Add a streamUrl (or episode url) "
                            + "from your licensed catalog API. See README.")
                    .setPositiveButton("OK", null)
                    .show();
            return;
        }
        Intent i = new Intent(this, PlayerActivity.class);
        i.putExtra("id", c.id);
        i.putExtra("season", season);
        i.putExtra("episode", episode);
        startActivity(i);
    }

package com.cinemabox.app;

import android.os.Bundle;
import android.view.KeyEvent;
import android.view.View;
import android.view.WindowManager;
import android.widget.TextView;

import androidx.appcompat.app.AppCompatActivity;

import com.cinemabox.app.data.Repository;
import com.cinemabox.app.data.UserStore;

import com.google.android.exoplayer2.MediaMetadata;
import com.google.android.exoplayer2.PlaybackException;
import com.google.android.exoplayer2.Player;
import com.google.android.exoplayer2.ui.StyledPlayerView;

import java.util.ArrayList;
import java.util.List;

/**
 * Player with Play/Pause, seek bar, speed, quality, audio and subtitle selection
 * (built into the ExoPlayer controller settings menu), next/previous episode,
 * resume playback, always fullscreen landscape.
 */
public class PlayerActivity extends AppCompatActivity {

    private StyledPlayerView playerView;
    private TextView messageView;
    private ExoPlayer player;
    private UserStore store;
    private Content c;
    private final List<MediaItem> items = new ArrayList<>();
    private int startIndex = 0;
    private long startPos = 0;
    private String currentMediaId;
    private boolean itemsReady = false, started = false;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        getWindow().addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON);
        setContentView(R.layout.activity_player);
        playerView = findViewById(R.id.playerView);
        messageView = findViewById(R.id.playerMessage);
        store = new UserStore(this);

        final String id = getIntent().getStringExtra("id");
        final int season = getIntent().getIntExtra("season", -1);
        final int episode = getIntent().getIntExtra("episode", 0);

        messageView.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                if (player != null) {
                    messageView.setVisibility(View.GONE);
                    player.prepare();
                }
            }
        });

        Repository.load(this, false, new Repository.Callback() {
            @Override
            public void onResult(List<Content> list, String msg) {
                if (isFinishing()) return;
                for (Content x : list) if (x.id.equals(id)) c = x;
                if (c == null) {
                    showMessage("Title not found");
                    return;
                }
                buildItems(season, episode);
                if (items.isEmpty()) {
                    showMessage("No authorized stream source for this title.");
                    return;
                }
                startPos = store.resumeEnabled() ? store.getPosition(items.get(startIndex).mediaId) : 0;
                itemsReady = true;
                if (started) initPlayer();
            }
        });
    }

    private void buildItems(int season, int episode) {
        if (c.hasEpisodes() && season >= 0 && season < c.seasons.size()) {
            Content.Season s = c.seasons.get(season);
            for (int i = 0; i < s.episodes.size(); i++) {
                Content.Episode e = s.episodes.get(i);
                if (e.url.isEmpty()) continue;
                if (i == episode) startIndex = items.size();
                items.add(item(e.id, e.url, c.title + " - " + e.title));
            }
        } else if (!c.streamUrl.isEmpty()) {
            items.add(item(c.id, c.streamUrl, c.title));
        }
    }

    private MediaItem item(String mediaId, String url, String title) {
        return new MediaItem.Builder()
                .setUri(url)
                .setMediaId(mediaId)
                .setMediaMetadata(new MediaMetadata.Builder().setTitle(title).build())
                .build();
    }

    private void initPlayer() {
        if (items.isEmpty() || player != null) return;
        player = new ExoPlayer.Builder(this).build();
        playerView.setPlayer(player);
        currentMediaId = items.get(startIndex).mediaId;
        player.setMediaItems(items, startIndex, startPos);
        player.addListener(new Player.Listener() {
            @Override
            public void onPlayerError(PlaybackException error) {
                showMessage("Playback error: " + error.getErrorCodeName() + "\nTap to retry");
            }

            @Override
            public void onMediaItemTransition(MediaItem item, int reason) {
                if (currentMediaId != null && reason == Player.MEDIA_ITEM_TRANSITION_REASON_AUTO) {
                    store.saveProgress(c.id, currentMediaId, 100, 100, c.hasEpisodes());
                }
                currentMediaId = item == null ? null : item.mediaId;
            }
        });
        messageView.setVisibility(View.GONE);
        player.setPlayWhenReady(true);
        player.prepare();
    }

    private void showMessage(String m) {
        messageView.setText(m);
        messageView.setVisibility(View.VISIBLE);
    }

    private void saveProgress() {
        if (player == null || c == null) return;
        MediaItem mi = player.getCurrentMediaItem();
        if (mi == null) return;
        long dur = player.getDuration();
        if (dur == C.TIME_UNSET) dur = 0;
        store.saveProgress(c.id, mi.mediaId, player.getCurrentPosition(), dur, c.hasEpisodes());
    }

    @Override
    protected void onStart() {
        super.onStart();
        started = true;
        if (itemsReady) initPlayer();
    }

    @Override
    protected void onPause() {
        super.onPause();
        if (player != null) player.pause();
    }

    @Override
    protected void onStop() {
        super.onStop();
        started = false;
        if (player != null) {
            saveProgress();
            startIndex = player.getCurrentMediaItemIndex();
            startPos = player.getCurrentPosition();
            playerView.setPlayer(null);
            player.release();
            player = null;
        }
    }

    @Override
    public boolean dispatchKeyEvent(KeyEvent event) {
        return playerView.dispatchKeyEvent(event) || super.dispatchKeyEvent(event);
    }

    @Override
    public void onWindowFocusChanged(boolean hasFocus) {
        super.onWindowFocusChanged(hasFocus);
        if (hasFocus) {
            getWindow().getDecorView().setSystemUiVisibility(
                    View.SYSTEM_UI_FLAG_LAYOUT_STABLE
                            | View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION
                            | View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN
                            | View.SYSTEM_UI_FLAG_HIDE_NAVIGATION
                            | View.SYSTEM_UI_FLAG_FULLSCREEN
                            | View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY);
        }
    }
}
